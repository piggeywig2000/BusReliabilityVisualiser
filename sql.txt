CREATE TABLE `bus_visualiser`.`trace_points` (
  `id` INT UNSIGNED NOT NULL AUTO_INCREMENT,
  `recorded_at` DATETIME NOT NULL,
  `vehicle_ref` VARCHAR(20) NOT NULL,
  `service_code` VARCHAR(40) NOT NULL,
  `line_id` VARCHAR(40) NOT NULL,
  `ticket_machine_service_code` VARCHAR(20) NOT NULL,
  `ticket_machine_journey_code` VARCHAR(20) NOT NULL,
  `easting` DOUBLE NOT NULL,
  `northing` DOUBLE NOT NULL,
  `bearing` DOUBLE NULL,
  PRIMARY KEY (`id`),
  UNIQUE INDEX `id_UNIQUE` (`id` ASC) INVISIBLE,
  INDEX `idx_recorded_at` (`recorded_at` ASC) VISIBLE,
  INDEX `idx_vehicle_ref` (`vehicle_ref` ASC) VISIBLE,
  INDEX `idx_service_line` (`service_code` ASC, `line_id` ASC) VISIBLE);

CREATE USER 'busvisualiserwebapp'@'localhost' IDENTIFIED BY 'password';
GRANT SELECT, INSERT ON `bus_visualiser`.`trace_points` TO 'busvisualiserwebapp'@'localhost';
FLUSH PRIVILEGES;
